class Solution {
    public int lengthOfLongestSubstring(String s) {
        if(s.length()==0) return 0;
        int l=0;
        int m=0;
        int r=0;
        while(l<s.length())
        {
        HashSet<String> set=new HashSet<>();
            while(r < s.length() && !set.contains(String.valueOf(s.charAt(r))))        {
            set.add(String.valueOf(s.charAt(r)));
            r++;
        }
        m=Math.max(m,Math.abs(r-l+1));
        l=r;
        }
        return m-1;
    }
}
